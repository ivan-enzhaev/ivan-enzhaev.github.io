Step 3: Create ozz Stream Wrapper (src/ozz_sdl_stream.h) Шаг 3: Создание обертки потока ozz (src/ozz_sdl_stream.h)

To let ozz-animation load binary archives via SDL3 file streams, create src/ozz_sdl_stream.h: Чтобы разрешить ozz-animation загружать бинарные архивы через файловые потоки SDL3, создайте src/ozz_sdl_stream.h: 

#pragma once

#include <SDL3/SDL.h>
#include <ozz/base/io/stream.h>

class OzzSDLStream : public ozz::io::Stream
{
public:
    OzzSDLStream(SDL_IOStream *io)
        : io_(io)
        , opened_(io != nullptr)
    {
        if (io_)
            SDL_SeekIO(io_, 0, SDL_IO_SEEK_SET);
    }

    ~OzzSDLStream()
    {
        if (io_)
            SDL_CloseIO(io_);
    }

    bool opened() const override { return opened_; }

    size_t Read(void *buffer, size_t size) override
    {
        size_t total = 0;
        while (total < size)
        {
            size_t read = SDL_ReadIO(io_, (uint8_t *)buffer + total, size - total);
            if (read == 0)
                break;
            total += read;
        }
        return total;
    }

    size_t Write(const void *, size_t) override { return 0; }

    int Seek(int offset, Origin origin) override
    {
        SDL_IOWhence whence = SDL_IO_SEEK_SET;
        if (origin == kCurrent)
            whence = SDL_IO_SEEK_CUR;
        else if (origin == kEnd)
            whence = SDL_IO_SEEK_END;

        Sint64 pos = SDL_SeekIO(io_, offset, whence);
        return pos >= 0 ? 0 : -1;
    }

    int Tell() const override
    {
        Sint64 pos = SDL_TellIO(io_);
        return pos >= 0 ? (int)pos : -1;
    }

    size_t Size() const override
    {
        Sint64 size = SDL_GetIOSize(io_);
        return size >= 0 ? (size_t)size : 0;
    }

private:
    SDL_IOStream *io_;
    bool opened_;
};

Step 5: Core Animation Loading & Sampling Snippet Шаг 5: Основной фрагмент загрузки и семплирования анимации

Inside your application initialization and iteration loop, load and sample your ozz files as follows: Внутри инициализации приложения и цикла итерации загружайте и семплируйте файлы ozz следующим образом: 

// Load Ozz Skeleton
const char *skeletonPath = "assets/models/y-bot/skeleton.ozz";
SDL_IOStream *skeletonIo = SDL_IOFromFile(skeletonPath, "rb");
if (skeletonIo)
{
    OzzSDLStream stream(skeletonIo);
    ozz::io::IArchive archive(&stream);
    if (archive.TestTag<ozz::animation::Skeleton>())
    {
        ozz::animation::Skeleton tempSkeleton;
        archive >> tempSkeleton;
        app->ozzSkeleton = std::move(tempSkeleton);
    }
    else
    {
        SDL_Log("CRITICAL: Archive tag mismatch for Skeleton!");
    }
}
else
{
    SDL_Log("Failed to open ozz skeleton: %s", skeletonPath);
    return SDL_APP_FAILURE;
}

// Load Ozz Animation
const char *animationPath = "assets/models/y-bot/animation.ozz";
SDL_IOStream *animIo = SDL_IOFromFile(animationPath, "rb");
if (animIo)
{
    OzzSDLStream stream(animIo);
    ozz::io::IArchive archive(&stream);
    archive >> app->ozzAnimation;
}
else
{
    SDL_Log("Failed to open ozz animation: %s", animationPath);
    return SDL_APP_FAILURE;
}

// Initialize sampling context and timer
app->ozzSamplingContext.Resize(app->ozzSkeleton.num_joints());
app->ozzAnimationTime = 0.0f;
app->ozzModelMatrices.resize(app->ozzSkeleton.num_joints());

// Inside frame update loop:
app->ozzAnimationTime += delta_time;
if (app->ozzAnimationTime > app->ozzAnimation.duration())
{
    app->ozzAnimationTime = fmod(app->ozzAnimationTime, app->ozzAnimation.duration());
}

// Setup sampling job
ozz::animation::SamplingJob sampling_job;
sampling_job.animation = &app->ozzAnimation;
sampling_job.ratio = app->ozzAnimation.duration() > 0.0f ? (app->ozzAnimationTime / app->ozzAnimation.duration()) : 0.0f;
sampling_job.context = &app->ozzSamplingContext;

ozz::vector<ozz::math::SoaTransform> locals;
locals.resize(app->ozzSkeleton.num_soa_joints());
sampling_job.output = make_span(locals);

if (!sampling_job.Run())
{
    SDL_Log("Failed to sample ozz animation at ratio: %f", sampling_job.ratio);
}

// Convert local transforms to model-space matrices
ozz::animation::LocalToModelJob conv_job;
conv_job.skeleton = &app->ozzSkeleton;
conv_job.input = make_span(locals);
conv_job.output = make_span(app->ozzModelMatrices);

if (!conv_job.Run())
{
    SDL_Log("Failed to run local-to-model animation job.");
}
