Project Configuration (CMakeLists.txt) Конфигурация проекта (CMakeLists.txt)

Copy and paste the following code into your CMakeLists.txt file. This configuration links SDL3, SDL3_ttf, Glad, ImGui, and your new ImPlot setup via MinGW static linking: Скопируйте и вставьте следующее содержимое в файл CMakeLists.txt. Эта конфигурация связывает SDL3, SDL3_ttf, Glad, ImGui и ImPlot с помощью статической линковки в MinGW:

CMakeLists.txt
cmake_minimum_required(VERSION 3.21)
project(sdl3-opengl-implot-cpp)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE "Debug")

# Main Executable
add_executable(app)

# Project and Library Sources
target_sources(app PRIVATE
    src/main.cpp
    "C:/libs/glad-0.1.36/opengl-3.3/src/glad.c"
    "C:/libs/imgui-1.92.6-mingw/src/imgui_impl_opengl3.cpp"
    "C:/libs/imgui-1.92.6-mingw/src/imgui_impl_sdl3.cpp"
    "C:/libs/implot-1.0-mingw/src/implot.cpp"
    "C:/libs/implot-1.0-mingw/src/implot_items.cpp"
    "C:/libs/implot-1.0-mingw/src/implot_demo.cpp"
)

# Include Directories
target_include_directories(app PRIVATE 
    "C:/libs/glad-0.1.36/opengl-3.3/include"
    "C:/libs/imgui-1.92.6-mingw/include"
    "C:/libs/implot-1.0-mingw/include"
    "C:/libs/cglm-0.9.6/include"
)

# Environmental Paths for MinGW Dependencies
if (WIN32)
    set(SDL3_DIR "C:/libs/SDL3-devel-3.4.8-mingw/lib/cmake/SDL3")
    set(SDL3_ttf_DIR "C:/libs/SDL3_ttf-devel-3.2.2-mingw/lib/cmake/SDL3_ttf")

    # Link directories for built static libs (if applicable)
    target_link_directories(app PRIVATE 
        "C:/libs/imgui-1.92.6-mingw/lib"
        "C:/libs/implot-1.0-mingw/lib"
    )

    # Static link options to prevent runtime DLL dependency issues
    target_link_options(app PRIVATE -static -mconsole)

    # Copy assets folder to output directory post-build
    add_custom_command(TARGET app POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:app>/assets
        COMMAND_EXPAND_LISTS
    )
endif()

# Find and Bind Dependencies
find_package(SDL3 REQUIRED)
find_package(SDL3_ttf REQUIRED)

target_link_libraries(app PRIVATE 
    SDL3_ttf::SDL3_ttf 
    SDL3::SDL3 
    ImGui
    ImPlot
)

Source Code Исходный код

Paste this code into src/main.cpp to initialize SDL3, OpenGL, ImGui, ImPlot and render a basic line graph: Вставьте этот код в src/main.cpp для инициализации SDL3, OpenGL, ImGui, ImPlot и вывода базового графика:

main.cpp
#include <glad/glad.h>
#include <SDL3/SDL.h>
#include <imgui.h>
#include <imgui_impl_sdl3.h>
#include <imgui_impl_opengl3.h>
#include <implot.h>
#include <cmath>

int main(int argc, char* argv[])
{
    if (!SDL_Init(SDL_INIT_VIDEO))
        return -1;

    SDL_GL_SetAttribute(SDL_GL_CONTEXT_PROFILE_MASK, SDL_GL_CONTEXT_PROFILE_CORE);
    SDL_GL_SetAttribute(SDL_GL_CONTEXT_MAJOR_VERSION, 3);
    SDL_GL_SetAttribute(SDL_GL_CONTEXT_MINOR_VERSION, 3);

    SDL_Window* window = SDL_CreateWindow("SDL3 + ImPlot Sandbox", 800, 600, SDL_WINDOW_OPENGL | SDL_WINDOW_RESIZABLE);
    SDL_GLContext glContext = SDL_GL_CreateContext(window);

    if (!gladLoadGLLoader((GLADloadproc)SDL_GL_GetProcAddress))
        return -1;

    SDL_GL_SetSwapInterval(1);

    // Setup Dear ImGui Context
    IMGUI_CHECKVERSION();
    ImGui::CreateContext();
    ImPlot::CreateContext(); // Crucial: Initialize ImPlot context!

    ImGuiIO& io = ImGui::GetIO(); (void)io;
    ImGui::StyleColorsDark();

    ImGui_implSDL3_InitForOpenGL(window, glContext);
    ImGui_ImplOpenGL3_Init("#version 330 core");

    // Sample data generation for ImPlot
    float x_data[100], y_data[100];
    for (int i = 0; i < 100; ++i) {
        x_data[i] = i * 0.1f;
        y_data[i] = std::sin(x_data[i]);
    }

    bool running = true;
    SDL_Event event;
    while (running)
    {
        while (SDL_PollEvent(&event))
        {
            ImGui_implSDL3_ProcessEvent(&event);
            if (event.type == SDL_EVENT_QUIT)
                running = false;
        }

        // Start ImGui Frame
        ImGui_ImplOpenGL3_NewFrame();
        ImGui_implSDL3_NewFrame();
        ImGui::NewFrame();

        // Render ImPlot Window
        ImGui::Begin("Plot Analytics");
        if (ImPlot::BeginPlot("Sine Wave Functional View")) {
            ImPlot::PlotLine("f(x) = sin(x)", x_data, y_data, 100);
            ImPlot::EndPlot();
        }
        ImGui::End();

        // Rendering Pipeline
        ImGui::Render();
        glViewport(0, 0, (int)io.DisplaySize.x, (int)io.DisplaySize.y);
        glClearColor(0.15f, 0.16f, 0.21f, 1.0f);
        glClear(GL_COLOR_BUFFER_BIT);
        
        ImGui_ImplOpenGL3_RenderDrawData(ImGui::GetDrawData());
        SDL_GL_SwapWindow(window);
    }

    // Cleanup Contexts
    ImGui_ImplOpenGL3_Shutdown();
    ImGui_implSDL3_Shutdown();
    ImPlot::DestroyContext();
    ImGui::DestroyContext();

    SDL_GL_DestroyContext(glContext);
    SDL_DestroyWindow(window);
    SDL_Quit();
    return 0;
}
