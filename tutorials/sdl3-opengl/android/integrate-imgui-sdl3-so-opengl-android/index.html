Integrating Dear ImGui with SDL3 for Android (NDK & CMake) Интеграция Dear ImGui с SDL3 для Android (NDK и CMake)

A step-by-step guide to configuring and integrating Dear ImGui with SDL3 shared library using CMake for Android on Windows using NDK. Пошаговое руководство по настройке и интеграции библиотеки Dear ImGui с динамической библиотекой SDL3 с использованием CMake для Android в Windows с использованием NDK.

1A. File Contents for main.cpp (Classic Loop) 1A. Содержимое файла main.cpp (Классический цикл) 

#define SDL_MAIN_HANDLED
#include <iostream>
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>

#include "imgui.h"
#include "imgui_impl_sdl3.h"

int main(int argc, char* argv[]) {
    // Notify SDL that the application handles the main entry point
    SDL_SetMainReady();

    // Use dummy video driver for headless ADB shell execution (no Java Activity)
    SDL_SetHint(SDL_HINT_VIDEO_DRIVER, "dummy");

    // 1. Initialize SDL3
    if (!SDL_Init(SDL_INIT_VIDEO | SDL_INIT_EVENTS)) {
        std::cerr << "SDL Initialization failed: " << SDL_GetError() << std::endl;
        return -1;
    }

    // 2. Create headless SDL3 Window
    SDL_Window* window = SDL_CreateWindow("Dear ImGui + SDL3 Test", 1280, 720, SDL_WINDOW_HIDDEN);
    if (!window) {
        std::cerr << "Window creation failed: " << SDL_GetError() << std::endl;
        SDL_Quit();
        return -1;
    }

    // 3. Setup Dear ImGui context
    IMGUI_CHECKVERSION();
    ImGui::CreateContext();
    ImGuiIO& io = ImGui::GetIO(); (void)io;
    io.ConfigFlags |= ImGuiConfigFlags_NavEnableKeyboard;

    // Set virtual display size and frame delta time for headless execution verification
    io.DisplaySize = ImVec2(1280.0f, 720.0f);
    io.DeltaTime = 1.0f / 60.0f;

    // Build font atlas explicitly for headless verification
    unsigned char* pixels;
    int width, height;
    io.Fonts->GetTexDataAsRGBA32(&pixels, &width, &height);

    ImGui::StyleColorsDark();

    // 4. Setup Platform backend (SDL3)
    ImGui_ImplSDL3_InitForOther(window);

    // 5. Simulate an ImGui frame lifecycle with SDL3
    ImGui_ImplSDL3_NewFrame();
    ImGui::NewFrame();

    ImGui::Begin("ImGui + SDL3 Android Test Window");
    ImGui::Text("Hello, Dear ImGui with SDL3 on Android!");
    ImGui::End();

    ImGui::Render();

    std::cout << "SDL3 version: " << SDL_GetVersion() << std::endl;
    std::cout << "Dear ImGui version: " << IMGUI_VERSION << std::endl;
    std::cout << "Dear ImGui + SDL3 Backend initialized and verified successfully on Android!" << std::endl;

    // 6. Cleanup
    ImGui_ImplSDL3_Shutdown();
    ImGui::DestroyContext();

    SDL_DestroyWindow(window);
    SDL_Quit();

    return 0;
}

1B. File Contents for main.cpp (SDL3 Callbacks API) 1B. Содержимое файла main.cpp (SDL3 Callbacks API) 

#define SDL_MAIN_HANDLED
#include <iostream>
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>

#include "imgui.h"
#include "imgui_impl_sdl3.h"

struct AppState {
    SDL_Window* window = nullptr;
};

SDL_AppResult SDL_AppInit(void** appstate, int argc, char* argv[]) {
    SDL_SetHint(SDL_HINT_VIDEO_DRIVER, "dummy");

    if (!SDL_Init(SDL_INIT_VIDEO | SDL_INIT_EVENTS)) {
        std::cerr << "SDL Initialization failed: " << SDL_GetError() << std::endl;
        return SDL_APP_FAILURE;
    }

    AppState* state = new AppState();
    *appstate = state;

    state->window = SDL_CreateWindow("Dear ImGui + SDL3 Callbacks Test", 1280, 720, SDL_WINDOW_HIDDEN);
    if (!state->window) {
        return SDL_APP_FAILURE;
    }

    IMGUI_CHECKVERSION();
    ImGui::CreateContext();
    ImGuiIO& io = ImGui::GetIO(); (void)io;
    io.DisplaySize = ImVec2(1280.0f, 720.0f);
    io.DeltaTime = 1.0f / 60.0f;

    unsigned char* pixels;
    int width, height;
    io.Fonts->GetTexDataAsRGBA32(&pixels, &width, &height);

    ImGui::StyleColorsDark();
    ImGui_ImplSDL3_InitForOther(state->window);

    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void* appstate, SDL_Event* event) {
    ImGui_ImplSDL3_ProcessEvent(event);
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void* appstate) {
    ImGui_ImplSDL3_NewFrame();
    ImGui::NewFrame();

    ImGui::Begin("ImGui + SDL3 Callbacks API (Android)");
    ImGui::Text("Hello, Dear ImGui with SDL3 Callbacks on Android!");
    ImGui::End();

    ImGui::Render();

    std::cout << "SDL3 version: " << SDL_GetVersion() << std::endl;
    std::cout << "Dear ImGui version: " << IMGUI_VERSION << std::endl;
    std::cout << "Dear ImGui + SDL3 Callbacks API verified successfully on Android!" << std::endl;

    // Complete verification test after one frame
    return SDL_APP_SUCCESS;
}

void SDL_AppQuit(void* appstate, SDL_AppResult result) {
    if (appstate) {
        auto* state = static_cast<AppState*>(appstate);

        ImGui_ImplSDL3_Shutdown();
        ImGui::DestroyContext();

        if (state->window) {
            SDL_DestroyWindow(state->window);
        }
        delete state;
    }
    SDL_Quit();
}

int main(int argc, char* argv[]) {
    SDL_SetMainReady();
    return SDL_EnterAppMainCallbacks(argc, argv, SDL_AppInit, SDL_AppIterate, SDL_AppEvent, SDL_AppQuit);
}
SDL3 Note: Headless ADB Shell Execution Заметка SDL3: Запуск без графического интерфейса в ADB Shell

Android applications rendering to the screen require an active Java SDLActivity. When executing native command-line binaries via adb shell without an Android Activity, setting SDL_HINT_VIDEO_DRIVER to "dummy" and defining SDL_MAIN_HANDLED allows SDL3 and Dear ImGui to initialize and run headless tests reliably without encountering JNI-related segmentation faults. Приложениям Android с экранным рендерингом требуется активная Java SDLActivity. При выполнении нативных бинарных файлов через консоль adb shell без наличия Android Activity установка SDL_HINT_VIDEO_DRIVER в "dummy" и использование SDL_MAIN_HANDLED позволяет надежно инициализировать и тестировать SDL3 и Dear ImGui без падений из-за отсутствия контекста JNI.

2. File Contents for CMakeLists.txt 2. Содержимое файла CMakeLists.txt 

cmake_minimum_required(VERSION 3.21)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Release")
endif()

project(imgui_sdl3_opengl_android_test CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# -----------------------------------------------------------------------------
# Paths
# -----------------------------------------------------------------------------
set(IMGUI_DIR "C:/libs/imgui-1.93.0")
set(SDL3_DIR  "C:/libs/sdl3-3.4.18-android/${ANDROID_ABI}/lib/cmake/SDL3")

# -----------------------------------------------------------------------------
# Locate Official SDL3 CMake Package
# -----------------------------------------------------------------------------
find_package(SDL3 REQUIRED CONFIG)

# -----------------------------------------------------------------------------
# ImGui Sources (including Demo Window and SDL3 Backend)
# -----------------------------------------------------------------------------
set(IMGUI_SOURCES
    ${IMGUI_DIR}/imgui.cpp
    ${IMGUI_DIR}/imgui_demo.cpp
    ${IMGUI_DIR}/imgui_draw.cpp
    ${IMGUI_DIR}/imgui_tables.cpp
    ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/backends/imgui_impl_sdl3.cpp
    ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
)

# -----------------------------------------------------------------------------
# Executable Target
# -----------------------------------------------------------------------------
add_executable(app
    main.cpp
    ${IMGUI_SOURCES}
)

# -----------------------------------------------------------------------------
# Include Directories
# -----------------------------------------------------------------------------
target_include_directories(app PRIVATE
    ${IMGUI_DIR}
    ${IMGUI_DIR}/backends
)

# -----------------------------------------------------------------------------
# Backend Selection & Definitions
# -----------------------------------------------------------------------------
target_compile_definitions(app PRIVATE
    IMGUI_IMPL_OPENGL_ES3
)

# -----------------------------------------------------------------------------
# Optimization & Symbol Stripping (Release Only)
# -----------------------------------------------------------------------------
if(CMAKE_BUILD_TYPE STREQUAL "Release" OR CMAKE_BUILD_TYPE STREQUAL "MinSizeRel")
    target_compile_options(app PRIVATE -Os)
    target_link_options(app PRIVATE -s)
endif()

# -----------------------------------------------------------------------------
# RPATH: Look for shared libraries in the same folder as the binary ($ORIGIN)
# -----------------------------------------------------------------------------
set_target_properties(app PROPERTIES
    BUILD_WITH_INSTALL_RPATH TRUE
    INSTALL_RPATH "\$ORIGIN"
)

# -----------------------------------------------------------------------------
# Platform Linking & Dependencies (SDL3 Shared Library)
# -----------------------------------------------------------------------------
target_link_libraries(app PRIVATE
    SDL3::SDL3-shared
    GLESv3
    log
    android
)

Step 3: Create Build & Execution Scripts Шаг 3: Создание скриптов сборки и запуска

Create batch scripts to configure, compile, and run the binary on an Android device or emulator via ADB: Создайте пакетные bat-скрипты для конфигурации, компиляции и запуска исполняемого файла на Android-устройстве или эмуляторе через ADB:

Windows (Batch Scripts)

config-android.bat

@echo off
set NDK=H:/AndroidSDK/ndk/28.2.13676358

cmake -G "MinGW Makefiles" ^
  -DCMAKE_TOOLCHAIN_FILE=%NDK%/build/cmake/android.toolchain.cmake ^
  -DANDROID_ABI=arm64-v8a ^
  -DANDROID_PLATFORM=android-21 ^
  -DCMAKE_BUILD_TYPE=Release ^
  -S . -B dist/android

build-android.bat

@echo off
cmake --build dist/android --config Release --parallel

run-android.bat

@echo off
set ADB=H:/AndroidSDK/platform-tools/adb.exe
set DEVICE_DIR=/data/local/tmp/imgui-sdl3-test
set SDL3_LIB=C:\libs\sdl3-3.4.18-android\arm64-v8a\lib\libSDL3.so

echo [*] Creating target directory on device...
%ADB% shell "mkdir -p %DEVICE_DIR%"

echo [*] Pushing SDL3 shared library...
%ADB% push "%SDL3_LIB%" %DEVICE_DIR%/

echo [*] Pushing executable...
%ADB% push dist/android/app %DEVICE_DIR%/

echo [*] Setting permissions and executing...
%ADB% shell "chmod +x %DEVICE_DIR%/app"
%ADB% shell "cd %DEVICE_DIR% && LD_LIBRARY_PATH=. ./app"

Make sure an Android device or emulator is connected (verify with adb devices), then run the scripts sequentially in Command Prompt: Убедитесь, что Android-устройство или эмулятор подключены (проверьте с помощью команды adb devices), затем последовательно запустите скрипты в командной строке: 

config-android.bat
build-android.bat
run-android.bat
