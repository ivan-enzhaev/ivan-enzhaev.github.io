How to Build and Run an IXWebSocket C++ Server on Ubuntu VPS & Windows Как собрать и запустить C++ сервер IXWebSocket на Ubuntu VPS и Windows

Ubuntu Prerequisite: To compile the server on Linux, build the static dependencies first: Building ZLIB, MBedTLS & IXWebSocket 12.0.1 Static Libraries on Ubuntu VPS. Предварительные требования для Ubuntu: Чтобы собрать сервер на Linux, сначала соберите статические библиотеки: Сборка статических библиотек ZLIB, MBedTLS и IXWebSocket 12.0.1 на Ubuntu VPS.
Windows Prerequisite (MinGW): Предварительные требования для Windows (MinGW):
Before building on Windows, ensure you have compiled all required static libraries using MinGW: Перед сборкой на Windows убедитесь, что вы собрали все необходимые статические библиотеки с помощью MinGW:
  • Building Zlib Static Library for Windows (MinGW & CMake)
  • Building Mbed TLS Static Library for Windows (MinGW & CMake)
  • Building IXWebSocket Static Library for Windows (MinGW & CMake)

This tutorial demonstrates how to build a high-performance C++ WebSocket server using IXWebSocket, static MBedTLS, and ZLIB on both Ubuntu VPS and Windows (MinGW). В этом руководстве показано, как собрать высокопроизводительный C++ WebSocket сервер с использованием IXWebSocket, статических библиотек MBedTLS и ZLIB как на Ubuntu VPS, так и на Windows (MinGW).

1. Server Source Code (src/main.cpp) 1. Исходный код сервера (src/main.cpp)

Create src/main.cpp with an echo WebSocket server implementation: Создайте файл src/main.cpp с реализацией эхо-сервера WebSocket: 

#include <ixwebsocket/IXNetSystem.h>
#include <ixwebsocket/IXWebSocketServer.h>
#include <iostream>

int main()
{
    // Initialize networking subsystem (required for Windows / Winsock)
    ix::initNetSystem();

    int port = 8080;
    std::string host = "0.0.0.0";
    ix::WebSocketServer server(port, host);

    server.setOnConnectionCallback(
        [](std::weak_ptr<ix::WebSocket> webSocketWeak,
           std::shared_ptr<ix::ConnectionState> connectionState)
        {
            auto webSocket = webSocketWeak.lock();
            if (!webSocket)
            {
                return;
            }

            webSocket->setOnMessageCallback(
                [webSocketWeak, connectionState](const ix::WebSocketMessagePtr& msg)
                {
                    auto ws = webSocketWeak.lock();
                    if (!ws)
                    {
                        return;
                    }

                    if (msg->type == ix::WebSocketMessageType::Open)
                    {
                        std::cout << "[Server] New Connection ID: " << connectionState->getId() << std::endl;
                        std::cout << "[Server] Uri: " << msg->openInfo.uri << std::endl;
                        std::cout << "[Server] Headers:" << std::endl;
                        for (const auto& [key, value] : msg->openInfo.headers)
                        {
                            std::cout << "  " << key << ": " << value << std::endl;
                        }
                    }
                    else if (msg->type == ix::WebSocketMessageType::Close)
                    {
                        std::cout << "[Server] Connection closed. Code: " << msg->closeInfo.code
                                  << " Reason: " << msg->closeInfo.reason << std::endl;
                    }
                    else if (msg->type == ix::WebSocketMessageType::Message)
                    {
                        std::cout << "[Server] Received: " << msg->str << std::endl;

                        // Echo response back to client
                        ws->send("Echo from server: " + msg->str);
                    }
                    else if (msg->type == ix::WebSocketMessageType::Error)
                    {
                        std::cerr << "[Server] Error: " << msg->errorInfo.reason << std::endl;
                    }
                });
        });

    auto res = server.listen();
    if (!res.first)
    {
        std::cerr << "[Server] Error listening on " << host << ":" << port << " - " << res.second << std::endl;
        ix::uninitNetSystem();
        return 1;
    }

    server.start();
    std::cout << "[Server] WebSocket server running on " << host << ":" << port << ". Press Enter to stop..." << std::endl;

    std::cin.get();

    server.stop();
    ix::uninitNetSystem();

    return 0;
}

2. CMake Build Setup (CMakeLists.txt) 2. Конфигурация сборки CMake (CMakeLists.txt)

Add the following cross-platform CMakeLists.txt targeting MinGW on Windows and system static library paths on Ubuntu Linux: Добавьте кроссплатформенный CMakeLists.txt с поддержкой MinGW на Windows и путей к статическим библиотекам на Ubuntu Linux: 

set(CMAKE_BUILD_TYPE "Debug")

cmake_minimum_required(VERSION 3.21)
project(ixwebsocket-server-example)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(PROJECT_SOURCES
    src/main.cpp
)

add_executable(app)
target_sources(app PRIVATE ${PROJECT_SOURCES})

if(WIN32)
    set(IXWEBSOCKET_DIR "C:/libs/ixwebsocket-12.0.1-mingw")
    set(MBEDTLS_DIR     "C:/libs/mbedtls-4.2.0-mingw")
    set(ZLIB_DIR        "C:/libs/zlib-1.3.2-mingw")

    target_include_directories(app PRIVATE
        ${IXWEBSOCKET_DIR}/include
        ${MBEDTLS_DIR}/include
        ${ZLIB_DIR}/include
    )

    add_library(IXWebSocket_Static STATIC IMPORTED)
    set_target_properties(IXWebSocket_Static PROPERTIES
        IMPORTED_LOCATION "${IXWEBSOCKET_DIR}/lib/libixwebsocket.a"
    )

    add_library(MBedTLS_Static STATIC IMPORTED)
    set_target_properties(MBedTLS_Static PROPERTIES
        IMPORTED_LOCATION "${MBEDTLS_DIR}/lib/libmbedtls.a"
    )

    add_library(MBedX509_Static STATIC IMPORTED)
    set_target_properties(MBedX509_Static PROPERTIES
        IMPORTED_LOCATION "${MBEDTLS_DIR}/lib/libmbedx509.a"
    )

    add_library(MBedCrypto_Static STATIC IMPORTED)
    set_target_properties(MBedCrypto_Static PROPERTIES
        IMPORTED_LOCATION "${MBEDTLS_DIR}/lib/libmbedcrypto.a"
    )

    add_library(ZLIB_Static STATIC IMPORTED)
    set_target_properties(ZLIB_Static PROPERTIES
        IMPORTED_LOCATION "${ZLIB_DIR}/lib/libzs.a"
    )

    target_link_libraries(app PRIVATE
        IXWebSocket_Static
        MBedTLS_Static
        MBedX509_Static
        MBedCrypto_Static
        ZLIB_Static
        ws2_32
        crypt32
        bcrypt
    )

    target_compile_options(app PRIVATE -Os)
    target_link_options(app PRIVATE -static -s -static-libgcc -static-libstdc++ -mconsole)

else() # Linux / Ubuntu VPS Target
    target_include_directories(app PRIVATE
        /usr/local/include
    )

    add_library(IXWebSocket_Static STATIC IMPORTED)
    set_target_properties(IXWebSocket_Static PROPERTIES
        IMPORTED_LOCATION "/usr/local/lib/libixwebsocket.a"
    )

    add_library(MBedTLS_Static STATIC IMPORTED)
    set_target_properties(MBedTLS_Static PROPERTIES
        IMPORTED_LOCATION "/usr/local/lib/libmbedtls.a"
    )

    add_library(MBedX509_Static STATIC IMPORTED)
    set_target_properties(MBedX509_Static PROPERTIES
        IMPORTED_LOCATION "/usr/local/lib/libmbedx509.a"
    )

    add_library(MBedCrypto_Static STATIC IMPORTED)
    set_target_properties(MBedCrypto_Static PROPERTIES
        IMPORTED_LOCATION "/usr/local/lib/libmbedcrypto.a"
    )

    add_library(ZLIB_Static STATIC IMPORTED)
    set_target_properties(ZLIB_Static PROPERTIES
        IMPORTED_LOCATION "/usr/local/lib/libz.a"
    )

    find_package(Threads REQUIRED)

    target_link_libraries(app PRIVATE
        IXWebSocket_Static
        MBedTLS_Static
        MBedX509_Static
        MBedCrypto_Static
        ZLIB_Static
        Threads::Threads
    )
endif()

3. Building and Running on Ubuntu VPS 3. Сборка и запуск на Ubuntu VPS

Create shell scripts in your project root to handle the build and execution workflow on Ubuntu: Создайте shell-скрипты в корне проекта для управления процессами сборки и запуска на Ubuntu:

config-exe.sh

#!/usr/bin/env bash
set -e
cmake -G Ninja -S . -B dist/linux

build-exe.sh

#!/usr/bin/env bash
set -e
cmake --build dist/linux

run-exe.sh

#!/usr/bin/env bash
set -e
./dist/linux/app

Grant execute permissions and run the build scripts: Предоставьте права на выполнение и запустите скрипты сборки: 

chmod +x config-exe.sh build-exe.sh run-exe.sh
./config-exe.sh
./build-exe.sh
./run-exe.sh

4. Building and Running on Windows (MinGW) 4. Сборка и запуск на Windows (MinGW)

To build and run the same server locally on Windows using MinGW, create batch scripts in your project root. Make sure you have previously built the required static libraries: Чтобы собрать и запустить этот же сервер локально на Windows с помощью MinGW, создайте batch-скрипты в корне проекта. Убедитесь, что вы предварительно собрали необходимые статические библиотеки:

  • Zlib Static Library (MinGW)
  • Mbed TLS Static Library (MinGW)
  • IXWebSocket Static Library (MinGW)

config-exe.bat

cmake -G Ninja -S . -B dist/win

build-exe.bat

cmake --build dist/win

run-exe.bat

.\dist\win\app.exe

Execute the batch scripts sequentially in your Windows Command Prompt or PowerShell: Последовательно запустите batch-скрипты в командной строке Windows (cmd) или PowerShell: 

config-exe.bat
build-exe.bat
run-exe.bat
