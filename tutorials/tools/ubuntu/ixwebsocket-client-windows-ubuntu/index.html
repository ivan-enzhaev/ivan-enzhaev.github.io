Building a Cross-Platform IXWebSocket C++ Client (Windows & Ubuntu) Создание кроссплатформенного C++ клиента IXWebSocket (Windows и Ubuntu)

Prerequisite: To test this client against a live endpoint, you need an active WebSocket server running on your local machine or VPS: How to Build and Run an IXWebSocket C++ Server on Ubuntu VPS. Предварительные требования: Для тестирования клиента вам потребуется активный WebSocket-сервер на локальном ПК или VPS: Как собрать и запустить C++ сервер IXWebSocket на Ubuntu VPS.

This guide covers writing and compiling a lightweight, cross-platform C++ WebSocket client using IXWebSocket, static MBedTLS, and ZLIB. The exact same source code compiles on both Windows (MinGW) and Ubuntu. В этом руководстве рассмотрено создание и компиляция кроссплатформенного WebSocket-клиента на C++ с использованием IXWebSocket, статической библиотеки MBedTLS и ZLIB. Одинаковый исходный код собирается как под Windows (MinGW), так и под Ubuntu.

1. Client Source Code (src/main.cpp) 1. Исходный код клиента (src/main.cpp)

Create src/main.cpp with an interactive WebSocket client implementation: Создайте файл src/main.cpp с интерактивной реализацией WebSocket-клиента: 

#include <ixwebsocket/IXNetSystem.h>
#include <ixwebsocket/IXWebSocket.h>
#include <iostream>
#include <string>
#include <thread>

int main()
{
    // Initialize network subsystem (required for Windows / Winsock)
    ix::initNetSystem();

    ix::WebSocket webSocket;

    // Target server URL (Replace with your VPS IP or domain)
    std::string url("ws://127.0.0.1:8080");
    webSocket.setUrl(url);

    std::cout << "[Client] Connecting to " << url << "..." << std::endl;

    // Setup incoming message callback
    webSocket.setOnMessageCallback([](const ix::WebSocketMessagePtr& msg) {
        if (msg->type == ix::WebSocketMessageType::Message)
        {
            std::cout << "\n[Server Response] " << msg->str << std::endl;
            std::cout << "> " << std::flush;
        }
        else if (msg->type == ix::WebSocketMessageType::Open)
        {
            std::cout << "\n[Client] Connected successfully!" << std::endl;
            std::cout << "[Client] Type a message and press Enter (or type 'quit' to exit):" << std::endl;
            std::cout << "> " << std::flush;
        }
        else if (msg->type == ix::WebSocketMessageType::Closed)
        {
            std::cout << "\n[Client] Connection closed. Code: " << msg->closeInfo.code
                      << " Reason: " << msg->closeInfo.reason << std::endl;
        }
        else if (msg->type == ix::WebSocketMessageType::Error)
        {
            std::cerr << "\n[Client] Error: " << msg->errorInfo.reason << std::endl;
        }
    });

    // Start WebSocket background thread
    webSocket.start();

    // Interactive console input loop
    std::string input;
    while (true)
    {
        std::getline(std::cin, input);

        if (input == "quit" || input == "exit")
        {
            break;
        }

        if (!input.empty())
        {
            webSocket.send(input);
        }
    }

    // Clean stop
    webSocket.stop();
    ix::uninitNetSystem();

    return 0;
}

2. CMake Build Setup (CMakeLists.txt) 2. Конфигурация сборки CMake (CMakeLists.txt)

The following CMakeLists.txt automatically configures static library paths depending on whether you build on Windows (via MinGW) or Ubuntu: Следующий CMakeLists.txt автоматически настраивает пути к статическим библиотекам в зависимости от ОС (Windows/MinGW или Ubuntu): 

set(CMAKE_BUILD_TYPE "Debug")

cmake_minimum_required(VERSION 3.21)
project(ixwebsocket-client-example)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(PROJECT_SOURCES
    src/main.cpp
)

add_executable(app)
target_sources(app PRIVATE ${PROJECT_SOURCES})

if(WIN32)
    set(IXWEBSOCKET_DIR "C:/libs/ixwebsocket-12.0.1-mingw")
    set(MBEDTLS_DIR     "C:/libs/mbedtls-4.2.0-mingw")
    set(ZLIB_DIR        "C:/libs/zlib-1.3.2-mingw")

    target_include_directories(app PRIVATE
        ${IXWEBSOCKET_DIR}/include
        ${MBEDTLS_DIR}/include
        ${ZLIB_DIR}/include
    )

    add_library(IXWebSocket_Static STATIC IMPORTED)
    set_target_properties(IXWebSocket_Static PROPERTIES
        IMPORTED_LOCATION "${IXWEBSOCKET_DIR}/lib/libixwebsocket.a"
    )

    add_library(MBedTLS_Static STATIC IMPORTED)
    set_target_properties(MBedTLS_Static PROPERTIES
        IMPORTED_LOCATION "${MBEDTLS_DIR}/lib/libmbedtls.a"
    )

    add_library(MBedX509_Static STATIC IMPORTED)
    set_target_properties(MBedX509_Static PROPERTIES
        IMPORTED_LOCATION "${MBEDTLS_DIR}/lib/libmbedx509.a"
    )

    add_library(MBedCrypto_Static STATIC IMPORTED)
    set_target_properties(MBedCrypto_Static PROPERTIES
        IMPORTED_LOCATION "${MBEDTLS_DIR}/lib/libmbedcrypto.a"
    )

    add_library(ZLIB_Static STATIC IMPORTED)
    set_target_properties(ZLIB_Static PROPERTIES
        IMPORTED_LOCATION "${ZLIB_DIR}/lib/libzs.a"
    )

    target_link_libraries(app PRIVATE
        IXWebSocket_Static
        MBedTLS_Static
        MBedX509_Static
        MBedCrypto_Static
        ZLIB_Static
        ws2_32
        crypt32
        bcrypt
    )

    target_compile_options(app PRIVATE -Os)
    target_link_options(app PRIVATE -static -s -static-libgcc -static-libstdc++ -mconsole)

else() # Ubuntu
    target_include_directories(app PRIVATE
        /usr/local/include
    )

    add_library(IXWebSocket_Static STATIC IMPORTED)
    set_target_properties(IXWebSocket_Static PROPERTIES
        IMPORTED_LOCATION "/usr/local/lib/libixwebsocket.a"
    )

    add_library(MBedTLS_Static STATIC IMPORTED)
    set_target_properties(MBedTLS_Static PROPERTIES
        IMPORTED_LOCATION "/usr/local/lib/libmbedtls.a"
    )

    add_library(MBedX509_Static STATIC IMPORTED)
    set_target_properties(MBedX509_Static PROPERTIES
        IMPORTED_LOCATION "/usr/local/lib/libmbedx509.a"
    )

    add_library(MBedCrypto_Static STATIC IMPORTED)
    set_target_properties(MBedCrypto_Static PROPERTIES
        IMPORTED_LOCATION "/usr/local/lib/libmbedcrypto.a"
    )

    add_library(ZLIB_Static STATIC IMPORTED)
    set_target_properties(ZLIB_Static PROPERTIES
        IMPORTED_LOCATION "/usr/local/lib/libz.a"
    )

    find_package(Threads REQUIRED)

    target_link_libraries(app PRIVATE
        IXWebSocket_Static
        MBedTLS_Static
        MBedX509_Static
        MBedCrypto_Static
        ZLIB_Static
        Threads::Threads
    )
endif()

3. Building and Running the Client 3. Сборка и запуск клиента

Windows (MinGW / Batch Scripts)

config-exe.bat

cmake -G Ninja -S . -B dist/win

build-exe.bat

cmake --build dist/win

run-exe.bat

.\dist\win\app.exe
Ubuntu (Bash Scripts)

config-exe.sh

#!/usr/bin/env bash
set -e
cmake -G Ninja -S . -B dist/ubuntu

build-exe.sh

#!/usr/bin/env bash
set -e
cmake --build dist/ubuntu

run-exe.sh

#!/usr/bin/env bash
set -e
./dist/ubuntu/app

Grant execute permissions on Ubuntu before running: Предоставьте права на выполнение в Ubuntu перед запуском: 

chmod +x config-exe.sh build-exe.sh run-exe.sh
./config-exe.sh
./build-exe.sh
./run-exe.sh
